clc;clear;

% x=linspace(0,16*T,N);

x=(0:N-1)/fs;

scatter(x,f,".k");
p=sum(f.^2)/N;
%2
noise=(sqrt(p)/100)*randn(1,1024);
plot(1:N,noise);
p_noise=sum(noise.^2)/N;
s_n=10*log10(p/p_noise)
%3

scatter(x,f+noise);
F=fft(f+noise)*2/N;
f_x=fs*(0:N/2)/(N);
mag_F=abs(F(1:N/2+1));
plot(f_x,10*log10(mag_F));

```matlab
%% 8
%1
clc;clear;
theta=pi/3;
e=0.01;
real_theta=acos((1+e)*cos(theta));
err=real_theta-theta;
%2
%若err大于0 cos为减函数 即theta>pi/3 theta=电压-电流
% 须使得theta减小，即使得更容性，需要延迟电压
%反之 err<0  delay I

if err > 0  
    disp("对电压采样进行延时");
elseif err < 0
    disp("对电流采样进行延时");
else 
    disp("均不需要进行延时");
end
delta=2*pi/32768;
N=ceil(abs(err)/delta);
disp(N)
```
